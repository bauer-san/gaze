# Core runtime dependencies (pinned deployment set).
#
# These pins are verified to resolve together. Two constraints drive them:
#   * mediapipe 0.10.20 requires numpy<2, so numpy stays on the 1.26 series.
#   * mediapipe depends on opencv-contrib-python. Do NOT also install
#     opencv-python: two OpenCV distributions in one environment install
#     competing copies of the `cv2` package and shadow each other.
#
# Optional hardware backends are deliberately not listed here so that a plain
# `pip install -r requirements.txt` always succeeds:
#   * Intel RealSense -> requirements-realsense.txt
#   * Kinect v1       -> libfreenect with Python bindings, built from source
#                        (there is no working `freenect` package on PyPI).
numpy==1.26.4
opencv-contrib-python==4.11.0.86
mediapipe==0.10.20
PyYAML==6.0.3
